function buildTemplateCsv(state: PusulaStateV3): Blob {
  const groups = groupTimetableLessons(state.timetable, state.slotsPerDay);
  const lines = ["Hafta;Ders;KazanÄ±m (AynÄ± haftada birden Ã§ok kazanÄ±m iÃ§in | kullanÄ±n)"];
  for (let w = 1; w <= 37; w++) {
    groups.forEach((g) => {
      const first = g.lessons[0];
      const existing = first ? (state.outcomes[makeCellKey(w, first.dayIndex, first.slotIndex)] || "").replace(/\r?\n+/g, " | ") : "";
      lines.push(`${w};"${g.displayTitle}";"${existing.replace(/"/g, '""')}"`);
    });
  }
  return new Blob(["\uFEFF" + lines.join("\r\n")], { type: "text/csv;charset=utf-8" });
}

export interface DistributeOptions {
  subject: string;
  sections: string[];
  weeklyOutcomes: Record<number, string[]>;
  mode: "spread" | "sequential";
  skipDayHolidays?: boolean;
}

export interface DistributeResult {
  nextState: PusulaStateV3;
  writtenCells: number;
  skippedHolidaySlots: number;
}

export function distributePlanToSections(
  state: PusulaStateV3,
  opts: DistributeOptions
): DistributeResult {
  const skipHol = opts.skipDayHolidays !== false;
  const groups = groupTimetableLessons(state.timetable, state.slotsPerDay);
  const targetSubj = opts.subject.trim().toLocaleLowerCase("tr-TR");
  const targetSections = new Set(opts.sections.map((s) => s.trim().toLocaleLowerCase("tr-TR")));

  const matchedGroups = groups.filter(
    (g) =>
      g.subject.toLocaleLowerCase("tr-TR") === targetSubj &&
      (targetSections.size === 0 || targetSections.has((g.section || "").toLocaleLowerCase("tr-TR")))
  );

  const nextOutcomes = { ...state.outcomes };
  const nextTimestamps = { ...state.outcomeUpdatedAt };
  const now = Date.now();
  let writtenCells = 0;
  let skippedHolidaySlots = 0;

  for (const [weekStr, items] of Object.entries(opts.weeklyOutcomes)) {
    const week = Number(weekStr);
    if (!Number.isInteger(week) || week < 1 || week > state.maxWeek) continue;
    const cleanItems = items.map((x) => x.trim()).filter(Boolean);
    if (cleanItems.length === 0) continue;

    for (const grp of matchedGroups) {
      const usableSlots: Array<{ dayIndex: number; slotIndex: number }> = [];
      for (const lesson of grp.lessons) {
        const dateIso = getLessonDateIso(state, week, lesson.dayIndex);
        if (skipHol && state.dayHolidays[dateIso]) {
          skippedHolidaySlots += 1;
          continue;
        }
        usableSlots.push(lesson);
      }
      if (usableSlots.length === 0) continue;

      const total = usableSlots.length;
      if (cleanItems.length <= total) {
        if (opts.mode === "spread") {
          for (let i = 0; i < total; i++) {
            const idx = Math.min(cleanItems.length - 1, Math.floor((i * cleanItems.length) / total));
            const key = makeCellKey(week, usableSlots[i].dayIndex, usableSlots[i].slotIndex);
            nextOutcomes[key] = cleanItems[idx];
            nextTimestamps[key] = now;
            writtenCells += 1;
          }
        } else {
          for (let i = 0; i < cleanItems.length; i++) {
            const key = makeCellKey(week, usableSlots[i].dayIndex, usableSlots[i].slotIndex);
            nextOutcomes[key] = cleanItems[i];
            nextTimestamps[key] = now;
            writtenCells += 1;
          }
        }
      } else {
        const buckets: string[][] = Array.from({ length: total }, () => []);
        cleanItems.forEach((item, idx) => {
          const bIdx = Math.min(total - 1, Math.floor((idx * total) / cleanItems.length));
          buckets[bIdx].push(item);
        });
        buckets.forEach((b, i) => {
          if (b.length === 0) return;
          const key = makeCellKey(week, usableSlots[i].dayIndex, usableSlots[i].slotIndex);
          nextOutcomes[key] = b.join("\n");
          nextTimestamps[key] = now;
          writtenCells += 1;
        });
      }
    }
  }

  return {
    nextState: {
      ...state,
      outcomes: nextOutcomes,
      outcomeUpdatedAt: nextTimestamps,
      updatedAt: now
    },
    writtenCells,
    skippedHolidaySlots
  };
}

          lastOutcomePreview = val.split("\n")[0].slice(0, 80);
        }
      }
    }

    const completionPct =
      totalActiveSlots > 0 ? Math.round((filledSlots / totalActiveSlots) * 100) : 0;

      groupKey: grp.groupKey,
      displayTitle: grp.displayTitle,
      weeklyHours: grp.lessons.length,
      totalActiveSlots,

      completionPct,

export function parseUtcDate(iso: string): Date {
  const [y, m, d] = iso.split("-").map(Number);
  return new Date(Date.UTC(y, m - 1, d));
}

export function formatUtcDate(date: Date): string {
  return (
    date.getUTCFullYear() +
    "-" +
    String(date.getUTCMonth() + 1).padStart(2, "0") +
    "-" +
    String(date.getUTCDate()).padStart(2, "0")
  );
}

export function toMondayIso(iso: string): string {
  const dt = parseUtcDate(iso);
  const day = dt.getUTCDay();
  dt.setUTCDate(dt.getUTCDate() + (day === 0 ? -6 : 1 - day));
  return formatUtcDate(dt);
}

export function addDaysIso(iso: string, days: number): string {
  const dt = parseUtcDate(iso);
  dt.setUTCDate(dt.getUTCDate() + days);

}

    return `Sonraki tatil: ${fmtShort(nextMon)} â€“ ${fmtFull(endFri)} (${base})`;

    let firstMon = prevMon;

  const t = new Uint32Array(256);
  for (let i = 0; i < 256; i++) {
    let c = i;
    for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;

function crc32(bytes: Uint8Array): number {
  let crc = 0xffffffff;
  for (let i = 0; i < bytes.length; i++) crc = CRC_TABLE[(crc ^ bytes[i]) & 0xff] ^ (crc >>> 8);
  return (crc ^ 0xffffffff) >>> 0;
}

function buildZipBlob(entries: Array<{ name: string; content: string }>, mimeType: string): Blob {

function colLetter(idx: number): string {
  let n = idx + 1;
  let s = "";
  while (n > 0) {
    s = String.fromCharCode(65 + ((n - 1) % 26)) + s;
    n = Math.floor((n - 1) / 26);
  }
  return s;
}

function xlsxCell(colIdx: number, rowNum: number, val: string, style: number): string {
  return `<c r="${colLetter(colIdx) + rowNum}" t="inlineStr" s="${style}"><is><t xml:space="preserve">${xmlEscape(val)}</t></is></c>`;
}

function wrapText(ctx: CanvasRenderingContext2D, text: string, maxW: number): string[] {
  const words = text.split(/\s+/).filter(Boolean);
  if (words.length === 0) return [];
  const lines: string[] = [];
  let cur = "";
  for (const w of words) {
    const test = cur ? `${cur} ${w}` : w;
    if (ctx.measureText(test).width <= maxW) cur = test;
    else {
      if (cur) lines.push(cur);
      cur = w;
    }
  }
  if (cur) lines.push(cur);
  return lines;
}

function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
  const rad = Math.min(r, w / 2, h / 2);
  ctx.beginPath();
  ctx.moveTo(x + rad, y);
  ctx.lineTo(x + w - rad, y);
  ctx.quadraticCurveTo(x + w, y, x + w, y + rad);
  ctx.lineTo(x + w, y + h - rad);
  ctx.quadraticCurveTo(x + w, y + h, x + w - rad, y + h);
  ctx.lineTo(x + rad, y + h);
  ctx.quadraticCurveTo(x, y + h, x, y + h - rad);
  ctx.lineTo(x, y + rad);
  ctx.quadraticCurveTo(x, y, x + rad, y);
  ctx.closePath();
}

interface CardRowLayout {
  num: number;
  hasContent: boolean;
  height: number;
  outcomeRows: Array<Array<{ code: string | null; codeW: number; text: string }>>;
}

interface CardLayout {
  group: LessonUnitGroup;
  pal: SubjectPalette;
  colW: number;
  headH: number;
  totalH: number;
  rows: CardRowLayout[];
}

function layoutColumns(
  ctx: CanvasRenderingContext2D,
  state: PusulaStateV3,
  week: number,
  groups: LessonUnitGroup[],
  scale: number
) {
  const margin = 22;
  const gap = 12;
  const cardGap = Math.max(8, Math.round(11 * scale));
  const numColW = Math.round(33 * scale);
  const padX = Math.round(9 * scale);
  const rowPadY = Math.max(5, Math.round(7 * scale));
  const lineH = Math.round(14 * scale * 1.32);
  const outGap = Math.max(3, Math.round(4 * scale));
  const codePadX = Math.round(4.5 * scale);
  const codeGap = Math.round(5.5 * scale);
  const subjFont = `700 ${Math.round(16 * scale)}px Georgia, serif`;
  const bodyFont = `400 ${Math.round(14 * scale)}px system-ui, sans-serif`;
  const codeFont = `700 ${Math.round(12 * scale)}px monospace`;
  const numFont = `700 ${Math.round(12.5 * scale)}px monospace`;

  const mainG = groups.filter((g) => subjectPriority(g.subject) === 0);
  const tailG = groups.filter((g) => subjectPriority(g.subject) > 0);
  const hasTail = tailG.length > 0;
  const totalCols = Math.min(5, Math.max(1, mainG.length + (hasTail ? 1 : 0)));
  const mainCols = hasTail && totalCols > 1 ? totalCols - 1 : totalCols;

  const ratios =
    totalCols === 5 ? [1.46, 0.98, 0.98, 0.98, 0.86] :
    totalCols === 4 ? [1.38, 0.98, 0.98, 0.9] :
    totalCols === 3 ? [1.3, 1, 0.92] :
    totalCols === 2 ? [1.22, 0.95] : [1];
  const rSum = ratios.reduce((a, b) => a + b, 0);
  const availW = 1920 - margin * 2 - gap * (totalCols - 1);
  const colWidths = ratios.map((r) => Math.floor((availW * r) / rSum));
  colWidths[0] += availW - colWidths.reduce((a, b) => a + b, 0);

  const colXs: number[] = [];
  let cx = margin;
  for (let i = 0; i < totalCols; i++) {
    colXs.push(cx);
    cx += colWidths[i] + gap;
  }

  const measureCard = (grp: LessonUnitGroup, colW: number): CardLayout => {
    const pal = getSubjectPalette(grp.subject);
    const innerW = colW - numColW - padX * 2;
    ctx.font = subjFont;
    const titleLines = wrapText(ctx, grp.displayTitle, colW - 88 * scale);
    const headH = Math.max(Math.round(35 * scale), titleLines.length * Math.round(19 * scale) + Math.round(16 * scale));

    const rows: CardRowLayout[] = grp.lessons.map((l, idx) => {
      const rawLines = (state.outcomes[makeCellKey(week, l.dayIndex, l.slotIndex)] || "")
        .trim()
        .split("\n")
        .map((x) => x.trim())
        .filter(Boolean);
      const outcomeRows: CardRowLayout["outcomeRows"] = [];
      let h = 0;

      if (rawLines.length === 0) {
        h = lineH;
      } else {
        rawLines.forEach((raw) => {
          const { code, rest } = splitOutcomeCode(raw);
          ctx.font = codeFont;
          const codeW = code ? ctx.measureText(code).width : 0;
          const codeBoxW = code ? codeW + codePadX * 2 + codeGap : 0;
          ctx.font = bodyFont;
          if (code && rest) {
            const words = rest.split(/\s+/).filter(Boolean);
            const firstLineWords: string[] = [];
            let built = "";
            const firstAvail = Math.max(60, innerW - codeBoxW);
            while (words.length > 0) {
              const test = built ? `${built} ${words[0]}` : words[0];
              if (ctx.measureText(test).width <= firstAvail) {
                built = test;
                firstLineWords.push(words.shift()!);
              } else break;
            }
            const segs: Array<{ code: string | null; codeW: number; text: string }> = [{ code, codeW, text: firstLineWords.join(" ") }];
            if (words.length > 0) {
              wrapText(ctx, words.join(" "), innerW).forEach((wLine) =>
                segs.push({ code: null, codeW: 0, text: wLine })
              );
            }
            outcomeRows.push(segs);
            h += segs.length * lineH + outGap;
          } else if (code) {
            outcomeRows.push([{ code, codeW, text: "" }]);
            h += lineH + outGap;
          } else {
            const wrapped = wrapText(ctx, raw, innerW);
            outcomeRows.push(wrapped.map((t) => ({ code: null, codeW: 0, text: t })));
            h += Math.max(1, wrapped.length) * lineH + outGap;
          }
        });
        h -= outGap;
      }
      return {
        num: idx + 1,
        hasContent: rawLines.length > 0,
        height: Math.max(Math.round(30 * scale), h + rowPadY * 2),
        outcomeRows
      };
    });

    const totalH = headH + rows.reduce((a, b) => a + b.height, 0);
    return { group: grp, pal, colW, headH, totalH, rows };
  };

  const columns: CardLayout[][] = Array.from({ length: totalCols }, () => []);
  const colHeights: number[] = Array.from({ length: totalCols }, () => 0);

  mainG.slice(0, mainCols).forEach((g, i) => {
    const card = measureCard(g, colWidths[i]);
    columns[i].push(card);
    colHeights[i] += card.totalH;
  });
  mainG.slice(mainCols).forEach((g) => {
    let best = mainCols > 1 ? 1 : 0;
    for (let i = mainCols > 1 ? 1 : 0; i < mainCols; i++) {
      if (colHeights[i] < colHeights[best]) best = i;
    }

function toUrlBase64(bytes: Uint8Array): string {
  let bin = "";
  for (let i = 0; i < bytes.length; i += 32768) {
    bin += String.fromCharCode(...bytes.subarray(i, i + 32768));
  }
  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

function fromUrlBase64(str: string): Uint8Array {
  const padded = str.replace(/-/g, "+").replace(/_/g, "/") + "===".slice((str.length + 3) % 4);
  const bin = atob(padded);
  const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

export async function decodeSyncCode(raw: string): Promise<PusulaStateV3> {
  const trimmed = raw.trim();
  if (!trimmed) throw new Error("EÅŸitleme kodu boÅŸ.");

  const match = trimmed.match(/(?:KD5Z|KD5|KD4Z|KD4):[A-Za-z0-9\-_]+/);
  let parsed: unknown;

  if (match) {
    const token = match[0];
    if (token.startsWith("KD5Z:") || token.startsWith("KD4Z:")) {
      if (typeof DecompressionStream === "undefined") {
        throw new Error("Bu tarayÄ±cÄ± sÄ±kÄ±ÅŸtÄ±rÄ±lmÄ±ÅŸ eÅŸitleme kodunu aÃ§mayÄ± desteklemiyor.");
      }
      const b64 = token.slice(5);
      const comp = fromUrlBase64(b64);
      const ds = new DecompressionStream("deflate-raw");
      const w = ds.writable.getWriter();
      await w.write(new Uint8Array(comp));
      await w.close();
      const text = await new Response(ds.readable).text();
      parsed = JSON.parse(text);
    } else {
      const b64 = token.slice(4);
      const text = new TextDecoder("utf-8").decode(fromUrlBase64(b64));
      parsed = JSON.parse(text);
    }
  } else {
    parsed = JSON.parse(trimmed);
  }

  const state = migrateV2ToV3(parsed);

}

export interface SyncMergeAnalysis {
  addedCount: number;
  keptLocalCount: number;
  conflictCount: number;
}

export function analyzeSyncMerge(local: PusulaStateV3, incoming: PusulaStateV3): SyncMergeAnalysis {
  const allKeys = new Set([...Object.keys(local.outcomes), ...Object.keys(incoming.outcomes)]);
  let addedCount = 0;
  let keptLocalCount = 0;
  let conflictCount = 0;

  allKeys.forEach((k) => {
    const a = (local.outcomes[k] || "").trim();
    const b = (incoming.outcomes[k] || "").trim();
    if (!a && b) addedCount++;
    else if (a && !b) keptLocalCount++;
    else if (a && b && a !== b) conflictCount++;
  });

  return { addedCount, keptLocalCount, conflictCount };
}

export function applySyncMerge(
  local: PusulaStateV3,
  incoming: PusulaStateV3,
  strategy: "smart" | "incoming" | "local" | "replace"
): PusulaStateV3 {
  if (strategy === "replace") return { ...incoming, updatedAt: Date.now() };

  const outcomes: Record<string, string> = {};
  const outcomeUpdatedAt: Record<string, number> = {};
  const now = Date.now();
  const allKeys = new Set([...Object.keys(local.outcomes), ...Object.keys(incoming.outcomes)]);

  allKeys.forEach((k) => {
    const a = (local.outcomes[k] || "").trim();
    const b = (incoming.outcomes[k] || "").trim();
    const ta = local.outcomeUpdatedAt[k] || 0;
    const tb = incoming.outcomeUpdatedAt[k] || 0;

    if (a && !b) {
      outcomes[k] = local.outcomes[k];
      if (ta > 0) outcomeUpdatedAt[k] = ta;
    } else if (!a && b) {
      outcomes[k] = incoming.outcomes[k];
      outcomeUpdatedAt[k] = tb > 0 ? tb : now;
    } else if (a && b) {
      const pickIncoming = a !== b && (strategy === "incoming" || (strategy === "smart" && tb > ta));
      outcomes[k] = pickIncoming ? incoming.outcomes[k] : local.outcomes[k];
      outcomeUpdatedAt[k] = Math.max(ta, tb) || now;
    }
  });

  const useIncomingMeta = strategy === "incoming" || (strategy === "smart" && incoming.updatedAt > local.updatedAt);
  return {
    ...(useIncomingMeta ? incoming : local),
    outcomes,
    outcomeUpdatedAt,
    teacherNotes: { ...incoming.teacherNotes, ...local.teacherNotes },
    updatedAt: Math.max(now, local.updatedAt, incoming.updatedAt)
  };
}
